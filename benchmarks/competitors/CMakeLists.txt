cmake_minimum_required(VERSION 3.19)
project(recode_comparison LANGUAGES C)
find_package(Recode 1.0 CONFIG REQUIRED)
option(COMPARISON_LLVM "Build the real LLVM ORC adapter" OFF)
option(COMPARISON_MIR "Build the real MIR adapter" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
function(comparison name)
    add_executable(${name} main.c workloads.c engine.c ${ARGN})
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(${name} PRIVATE Recode::recode)
endfunction()
comparison(compare)
if(COMPARISON_LLVM)
    find_program(LLVM_CONFIG llvm-config REQUIRED)
    execute_process(COMMAND ${LLVM_CONFIG} --includedir OUTPUT_VARIABLE LLVM_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND ${LLVM_CONFIG} --libdir OUTPUT_VARIABLE LLVM_LIBDIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    find_library(LLVM_LIBRARY NAMES LLVM HINTS ${LLVM_LIBDIR} REQUIRED)
    comparison(compare-llvm llvm.c)
    target_include_directories(compare-llvm SYSTEM PRIVATE ${LLVM_INCLUDE})
    target_link_libraries(compare-llvm PRIVATE ${LLVM_LIBRARY})
    target_compile_definitions(compare-llvm PRIVATE HAVE_LLVM)
endif()
if(COMPARISON_MIR)
    find_path(MIR_INCLUDE_DIR mir.h REQUIRED)
    find_library(MIR_LIBRARY NAMES mir mir_static REQUIRED)
    comparison(compare-mir mir.c)
    target_include_directories(compare-mir SYSTEM PRIVATE ${MIR_INCLUDE_DIR})
    target_link_libraries(compare-mir PRIVATE ${MIR_LIBRARY} m ${CMAKE_DL_LIBS})
    target_compile_definitions(compare-mir PRIVATE HAVE_MIR)
endif()
add_executable(lifecycle lifecycle.c)
target_compile_options(lifecycle PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(lifecycle PRIVATE Recode::recode)
include(CTest)
if(BUILD_TESTING)
    foreach(kind RANGE 0 5)
        foreach(mode IN ITEMS c interpret fixed adaptive)
            add_test(NAME ${mode}-${kind} COMMAND compare ${mode} ${kind} 100 1)
        endforeach()
        if(COMPARISON_LLVM)
            add_test(NAME llvm-${kind} COMMAND compare-llvm llvm ${kind} 100 1)
        endif()
        if(COMPARISON_MIR)
            add_test(NAME mir-${kind} COMMAND compare-mir mir ${kind} 100 1)
        endif()
    endforeach()
    add_test(NAME registration-disposal COMMAND lifecycle)
endif()
